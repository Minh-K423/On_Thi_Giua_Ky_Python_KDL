Đây là chữ ký thật của hai hàm sẽ được gọi hằng ngày trong học phần:

def tach_ban_ghi(dong):
"""Doi mot dong thanh (ten, mon, diem); tra None neu dong hong."""
phan = dong.split(",")
if len(phan) != 3:
return None
ten, mon, diem = phan
if ten == "" or mon == "" or diem == "":
return None
return ten, mon, float(diem)
