class SinhVien:
"""Sinh vien va cac diem."""
def __init__(self, ten, lop):
self.ten = ten
self.lop = lop
self.diem = [] # list rieng
def them_diem(self, d):
self.diem.append(d)
def diem_tb(self):
if not self.diem:
return 0.0
return sum(self.diem) / len(self.diem)
def xep_loai(self, nguong=5.0):
if self.diem_tb() >= nguong:
return "Dat"
return "Chua dat"

class ThongKeChay:
"""Thong ke tren dong so."""
def __init__(self):
self.n = 0
self.tong = 0.0
self.nho_nhat = None
self.lon_nhat = None
def them(self, x):
self.n += 1
self.tong += x
if self.n == 1 or x < self.nho_nhat:
self.nho_nhat = x
if self.n == 1 or x > self.lon_nhat:
self.lon_nhat = x
def trung_binh(self):
return self.tong / self.n
None nghĩa là “chưa có” . Số đầu tiên (self.n == 1 ) luôn thành nhỏ nhất và lớn nhất, nên không bao
giờ phải so sánh với None.

class MoHinhTrungBinh:
"""Du doan moi gia tri bang trung binh cua du lieu huan luyen."""
def __init__(self, lam_tron=2):
self.lam_tron = lam_tron # cau hinh: dat luc tao
def fit(self, y):
self.trung_binh_ = round(sum(y) / len(y), self.lam_tron)
return self # tra lai chinh no
def predict(self, n):
return [self.trung_binh_] * n
