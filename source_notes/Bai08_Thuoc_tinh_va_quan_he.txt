@classmethod: một cách tạo đối tượng khác
class SinhVien:
def __init__(self, ten, lop, diem=None):
self.ten, self.lop = ten, lop
self.diem = [] if diem is None else list(diem)
@classmethod
def tu_dong(cls, dong): # "ten;lop;diem,diem"
ten, lop, ds = dong.split(";")
return cls(ten, lop, [float(x) for x in ds.split(",")])
sv = SinhVien.tu_dong("Le Van Chi;24TTH;8.5,7.0")
print(sv.ten, sv.lop, sv.diem)
